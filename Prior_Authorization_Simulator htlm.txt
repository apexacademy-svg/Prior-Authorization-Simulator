  <script>
    // Initialize Lucide
    lucide.createIcons();

    // State
    let teacherMode = false;
    let documentationAttached = false;
    let auditEntries = [];

    // Config object for Element SDK
    let config = {
      app_title: "Prior Authorization Simulator",
      patient_name: "John Smith",
      provider_note: "Patient presents with chronic lower back pain and radiculopathy in left leg for 6+ months. Conservative treatment failed. MRI imaging of lumbar spine indicated to assess for disc herniation and nerve compression. Medical necessity: Rule out surgical pathology."
    };

    // Element SDK initialization
    async function initElementSDK() {
      if (window.elementSdk) {
        const defaultConfig = { ...config };
        
        window.elementSdk.init({
          defaultConfig,
          onConfigChange: async (newConfig) => {
            config = newConfig;
            document.getElementById('appTitle').textContent = config.app_title || defaultConfig.app_title;
            document.getElementById('emrPatientName').textContent = config.patient_name || defaultConfig.patient_name;
            document.getElementById('emrNotes').value = config.provider_note || defaultConfig.provider_note;
            lucide.createIcons();
          },
          mapToCapabilities: (cfg) => ({
            recolorables: [],
            borderables: [],
            fontEditable: undefined,
            fontSizeable: undefined
          }),
          mapToEditPanelValues: (cfg) => new Map([
            ["app_title", cfg.app_title || defaultConfig.app_title],
            ["patient_name", cfg.patient_name || defaultConfig.patient_name],
            ["provider_note", cfg.provider_note || defaultConfig.provider_note]
          ])
        });
      }
    }

    // Utility: Add audit log entry
    function addAuditEntry(message) {
      const time = new Date().toLocaleTimeString();
      const entry = `[${time}] ${message}`;
      auditEntries.push(entry);
      const auditLog = document.getElementById('auditLog');
      auditLog.innerHTML = auditEntries.map(e => `<p>${e}</p>`).join('');
      auditLog.scrollTop = auditLog.scrollHeight;
    }

    // Utility: Generate Auth Number
    function generateAuthNumber() {
      return 'AUTH-' + Math.floor(Math.random() * 100000).toString().padStart(5, '0');
    }

    // Teacher Mode Toggle
    document.getElementById('teacherBtn').addEventListener('click', () => {
      teacherMode = !teacherMode;
      const btn = document.getElementById('teacherBtn');
      const highlights = document.querySelectorAll('.teacher-highlight');
      
      if (teacherMode) {
        btn.classList.remove('bg-amber-500', 'hover:bg-amber-600');
        btn.classList.add('bg-green-600', 'hover:bg-green-700');
        highlights.forEach(el => {
          el.classList.add('ring-2', 'ring-green-500', 'bg-green-50');
        });
        addAuditEntry('✓ TEACHER MODE ACTIVATED - Key fields highlighted in green');
      } else {
        btn.classList.remove('bg-green-600', 'hover:bg-green-700');
        btn.classList.add('bg-amber-500', 'hover:bg-amber-600');
        highlights.forEach(el => {
          el.classList.remove('ring-2', 'ring-green-500', 'bg-green-50');
        });
        addAuditEntry('Teacher mode deactivated');
      }
    });

    // Check Eligibility
    document.getElementById('checkEligBtn').addEventListener('click', () => {
      const memberId = document.getElementById('eligibilityMemberID').value.trim();
      if (!memberId) {
        alert('Please enter a Member ID');
        return;
      }
      document.getElementById('eligibilityResult').classList.remove('hidden');
      addAuditEntry(`✓ Eligibility checked for Member ID: ${memberId}`);
    });

    // Attach Documentation
    document.getElementById('attachDocBtn').addEventListener('click', () => {
      documentationAttached = !documentationAttached;
      const btn = document.getElementById('attachDocBtn');
      const status = document.getElementById('attachStatus');
      if (documentationAttached) {
        btn.classList.remove('bg-blue-100', 'hover:bg-blue-200', 'text-blue-700', 'border-blue-400');
        btn.classList.add('bg-emerald-100', 'hover:bg-emerald-200', 'text-emerald-700', 'border-emerald-400');
        status.textContent = '✓ Documentation Attached';
        addAuditEntry('✓ Clinical documentation attached to request');
      } else {
        btn.classList.remove('bg-emerald-100', 'hover:bg-emerald-200', 'text-emerald-700', 'border-emerald-400');
        btn.classList.add('bg-blue-100', 'hover:bg-blue-200', 'text-blue-700', 'border-blue-400');
        status.textContent = 'Attach Clinical Documentation';
        addAuditEntry('Documentation removed from request');
      }
    });

    // Submit Authorization
    document.getElementById('submitBtn').addEventListener('click', () => {
      const npi = document.getElementById('npiField').value.trim();
      const icd10 = document.getElementById('icd10Field').value.trim();
      const cpt = document.getElementById('cptField').value.trim();
      const pos = document.getElementById('posField').value;

      if (!npi || !icd10 || !cpt) {
        alert('Please fill in all required fields (NPI, ICD-10, CPT)');
        return;
      }

      const resultCard = document.getElementById('resultCard');
      const resultStatus = document.getElementById('resultStatus');
      const resultMessage = document.getElementById('resultMessage');
      const resultAuthNumber = document.getElementById('resultAuthNumber');
      const resultIcon = document.getElementById('resultIcon');

      // Logic: Determine outcome based on CPT code
      let status, message, icon, authNum;

      if (cpt === '99213') {
        // Office Visit - No Auth Required
        status = 'No Auth Required';
        message = 'This procedure code does not require prior authorization for this member.';
        icon = 'info';
        authNum = 'N/A';
        resultCard.classList.remove('border-amber-300', 'bg-amber-50');
        resultCard.classList.add('border-blue-300', 'bg-blue-50');
      } else if (cpt === '72148' || cpt === '70553' || cpt === '70450') {
        // MRI/CT - Pending Review
        status = 'Pending - Review Required';
        message = 'Your request requires clinical review. Authorization pending.';
        icon = 'clock';
        authNum = generateAuthNumber();
        resultCard.classList.remove('border-amber-300', 'bg-amber-50');
        resultCard.classList.add('border-amber-300', 'bg-amber-50');
      } else {
        // Unknown CPT - Default Pending
        status = 'Pending - Review Required';
        message = 'Your request has been submitted for review.';
        icon = 'clock';
        authNum = generateAuthNumber();
        resultCard.classList.remove('border-amber-300', 'bg-amber-50');
        resultCard.classList.add('border-amber-300', 'bg-amber-50');
      }

      resultStatus.textContent = status;
      resultMessage.textContent = message;
      resultAuthNumber.textContent = authNum;
      resultIcon.setAttribute('data-lucide', icon);
      resultCard.classList.remove('hidden');
      lucide.createIcons();

      // Update EMR if auth generated
      if (authNum !== 'N/A') {
        document.getElementById('emrAuthNumber').value = authNum;
      }

      addAuditEntry(`✓ Authorization request submitted | CPT: ${cpt} | ICD-10: ${icd10} | NPI: ${npi} | Status: ${status}`);
    });

    // Copy Auth to Clipboard
    document.getElementById('copyAuthBtn').addEventListener('click', async () => {
      const authNum = document.getElementById('emrAuthNumber').value;
      if (!authNum) {
        alert('No authorization number to copy');
        return;
      }
      try {
        await navigator.clipboard.writeText(authNum);
        addAuditEntry(`✓ Authorization number copied to clipboard: ${authNum}`);
        const btn = document.getElementById('copyAuthBtn');
        btn.classList.add('pulse-animate');
        setTimeout(() => btn.classList.remove('pulse-animate'), 2000);
      } catch (err) {
        alert('Failed to copy');
      }
    });

    // Reset
    document.getElementById('resetBtn').addEventListener('click', () => {
      auditEntries = [];
      document.getElementById('auditLog').innerHTML = '<p class="text-gray-500">[System Ready] Waiting for user input...</p>';
      document.getElementById('emrAuthNumber').value = '';
      document.getElementById('resultCard').classList.add('hidden');
      document.getElementById('eligibilityResult').classList.add('hidden');
      document.getElementById('npiField').value = '1122334455';
      document.getElementById('icd10Field').value = 'M54.5';
      document.getElementById('cptField').value = '72148';
      document.getElementById('posField').value = '11';
      documentationAttached = false;
      document.getElementById('attachDocBtn').classList.remove('bg-emerald-100', 'hover:bg-emerald-200', 'text-emerald-700', 'border-emerald-400');
      document.getElementById('attachDocBtn').classList.add('bg-blue-100', 'hover:bg-blue-200', 'text-blue-700', 'border-blue-400');
      document.getElementById('attachStatus').textContent = 'Attach Clinical Documentation';
      addAuditEntry('System reset complete');
    });

    // Initialize
    initElementSDK();
    addAuditEntry('Prior Authorization Simulator loaded');
  </script>
